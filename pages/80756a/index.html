关键字

# 1.存储类型关键字

# 1.1 auto - 默认的局部变量存储类型

  • 局部变量默认存储类型，一般省略不写
void func() {
    auto int a = 10;  // 一般省略 auto，等同于 int a = 10;
    printf("%d\n", a);
}

# 1.2 register - 寄存器变量存储类型

  • 给编译器提建议，但是编译器可以不采纳，不能对register变量使用取地址运算符&，适用于局部变量
  • 编译器越来越聪明，所以正在逐渐废弃
void func() {
    register int i = 0;  // 建议将 i 存储在寄存器中（非强制）（加快访问速度）
}

# 1.3 static - 静态变量 / 静态函数

  • 局部静态变量：生命周期延长为整个程序期间
  • 限制作用域：用于函数或变量，限制其只在当前文件可见
void counter() {
    static int count = 0;
    count++;
    printf("Count: %d\n", count);
}
// 多次调用 counter 会保留 count 的值

static int globalVar = 10;  // 只在当前文件可见

# 1.4 extern - 外部引用声明

  • 不分配内存，只声明
  • 作用于跨文件访问全局变量/函数
// a.c 文件
int num = 100;

// b.c 文件
extern int num;  // 引用外部变量
printf("%d\n", num);

# 2.数据类型关键字

# 2.1 char - 1字节

char ch = 'A';
printf("Char: %c\n", ch);

# 2.2 int - 4字节

  • 某些平台上，int 可能是 2 字节
  • 不确定就用 sizeof(int) 计算
int x = 42;
printf("Int: %d\n", x);

# 2.3 float - 4字节

float pi = 3.14f;
printf("Pi: %f\n", pi);

# 2.4 double - 8字节

double d = 3.1415926;
printf("Double: %.6lf\n", d);

# 2.5 short - 2字节

short s = 100;
printf("Short: %d\n", s);

# 2.6 long - 4字节

  • 在 Windows（即使是 64 位） 上，long 仍然是 4 字节
  • 在 Linux/macOS 64 位 上，long 是 8 字节。
long l = 100000L;
printf("Long: %ld\n", l);

# 2.7 signed - 允许负数

  • 十进制负数 → 补码 正数原码 → 取反 + 1 = 补码
  • 补码 → 十进制负数 补码 → 取反 + 1 → 加负号
signed int a = -20;
printf("Signed: %d\n", a);

# 2.8 unsigned - 不允许负数

unsigned int b = 20;
printf("Unsigned: %u\n", b);

# 2.9 void - 无类型

void sayHello() {
    printf("Hello!\n");
}

int main() {
    sayHello();
    return 0;  // return 后的值可选
}

# 3.控制结构关键字

# 3.1 if / else

int a = 10;
if (a > 5) {
    printf("Greater than 5\n");
} else {
    printf("Less than or equal to 5\n");
}

# 3.2 switch / case / default

int x = 2;
switch (x) {
    case 1: printf("One\n"); break;
    case 2: printf("Two\n"); break;
    default: printf("Other\n"); break;
}

# 3.3 goto

int i = 0;
if (i == 0) goto label;
printf("Won't run\n");
label:
printf("Jumped here\n");

# 3.4 return

int add(int a, int b) {
    return a + b;
}

# 3.5 break

for (int i = 0; i < 10; i++) {
    if (i == 5) break;
    printf("%d ", i);
}

# 3.6 continue

for (int i = 0; i < 5; i++) {
    if (i == 2) continue;
    printf("%d ", i);  // 跳过 2
}

# 4.循环关键字

# 4.1 for

for (int i = 0; i < 3; i++) {
    printf("i = %d\n", i);
}

# 4.2 while

int i = 0;
while (i < 3) {
    printf("i = %d\n", i);
    i++;
}

# 4.3 do

int i = 0;
do {
    printf("i = %d\n", i);
    i++;
} while (i < 3);

# 5.结构/枚举/联合关键字

# 5.1 struct - 结构体

  • 用于将不同类型的变量组合成一个复合数据类型。
  • 结构体变量包含多个字段（成员）
  • 可配合 typedef 简化用法
struct Point {
    int x;
    int y;
};

struct Point p = {1, 2};
printf("x=%d, y=%d\n", p.x, p.y);

# 5.2 union - 共用体（联合体）

  • union（联合体）是一种复合数据类型，它可以包含多个成员，但同一时间只能使用其中一个成员，所有成员共享同一块内存空间。
  • 共享内存 所有成员使用同一块内存，只占用最大成员所需的空间
  • 节省空间 适合用在只会用一个字段的场景，比如协议、设备数据包等
  • 只能一个有效 同一时间只能访问一个字段，访问其他字段会导致不确定结果
union Data {
    int i;
    float f;
};

union Data d;
d.i = 10;
printf("i = %d\n", d.i);

# 5.3 enum - 枚举类型

  • 定义一组有名字的整型常量，提高代码可读性。
  • 默认从 0 开始递增, 可手动指定值
enum Color { RED, GREEN, BLUE };
enum Color c = GREEN;
printf("Color: %d\n", c); // 1

enum Week { MON = 1, TUE, WED }
enum Week today = TUE;
printf("%d\n", today);  // 2

# 6.其他关键字

# 6.1 typedef - 类型别名

  • 为复杂类型取别名（如结构体、函数指针）
  • 简化类型声明
typedef unsigned int uint;
uint a = 10;
printf("a = %u\n", a);

typedef struct Point{
    int x;
    int y;
} P;

P p1 = {1, 2};

# 6.2 sizeof - 计算类型或变量所占内存大小（单位：字节）

  • 编译时计算
  • 返回 size_t 类型
int x;
printf("%zu\n", sizeof(x));      // 4
printf("%zu\n", sizeof(int));    // 4

# 6.3 const - 只读常量

  • 声明后不能修改
  • 可配合指针限制指向或内容不可变
const int MAX = 100;
// MAX = 50; // 错误：不能修改

# 6.4 volatile - 易变变量（告诉编译器不要优化）

  • 多线程编程（变量被另一个线程修改）
  • 设备寄存器（如单片机）
  • 信号处理函数中的变量
volatile int status = 0;
// 编译器不会优化掉对 status 的重复访问（如用于硬件寄存器）

volatile int running = 1;

while (running) {
    // 可能在其他线程中将 running 设为 0
}
